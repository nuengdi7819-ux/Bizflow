📂 นำเข้าข้อมูลจากไฟล์สำรอง CSV (Restore Backup)
""" @app.route('/import_csv', methods=['POST']) def import_csv(): if 'admin' not in session: return redirect(url_for('login')) file = request.files.get('file') if file and file.filename.endswith('.csv'): try: # อ่านไฟล์ CSV ด้วย utf-8-sig เพื่อป้องกันสระเพี้ยน stream = io.StringIO(file.stream.read().decode("utf-8-sig"), newline=None) csv_input = csv.DictReader(stream) count_success = 0 for row in csv_input: customer_name = row.get('CustomerName') if not customer_name or not customer_name.strip(): continue # ฟังก์ชันแปลงค่าตัวเลขแบบปลอดภัย def safe_float(val, default=0.0): try: return float(val) if val and str(val).strip() else default except: return default # ฟังก์ชันแปลงค่าวันที่แบบปลอดภัย (ถ้าว่างหรือเป็น ### ให้เป็น None) def safe_date(val): try: val_str = str(val).strip() if not val_str or '###' in val_str or val_str.lower() == 'nan': return None return datetime.strptime(val_str, '%Y-%m-%d').date() except: return None # ฟังก์ชันแปลงค่า True/False def safe_bool(val): if not val: return False return str(val).strip().upper() == 'TRUE' # ตรวจสอบและจัดการเบอร์โทร (ถ้าเป็นค่าว่างหรือ nan ให้เป็น None) phone_val = row.get('Phone', '') if not phone_val or str(phone_val).strip().lower() == 'nan': phone_val = None else: phone_val = str(phone_val).strip() # ตรวจสอบเบอร์โทรกรณีติดเลขทศนิยม (เช่น float จาก pandas) if phone_val and '.' in phone_val: phone_val = phone_val.split('.')[0] p_val = safe_float(row.get('Principal')) orig_p = safe_float(row.get('OriginalPrincipal'), p_val) daily_int = safe_float(row.get('DailyInterest')) # สร้าง Transaction object ตามโครงสร้างฟิลด์ใน CSV new_tx = Transaction( type=row.get('Type', 'เงินกู้ยืม'), customer_name=customer_name.strip(), phone=phone_val, sales_name=row.get('SalesName', session.get('admin', 'nueng')), start_date=safe_date(row.get('StartDate')) or get_thai_today(), closed_date=safe_date(row.get('ClosedDate')), original_principal=orig_p, principal=p_val, daily_interest=daily_int, initial_daily_interest=daily_int, # ใช้ค่าเริ่มต้นเท่ากับ daily_interest paid_interest=safe_float(row.get('PaidInterest')), status=row.get('Status', 'ปกติ'), installment_amount=safe_float(row.get('InstallmentAmount')), schedule_type=row.get('ScheduleType', 'จ่ายทุกวัน'), due_day_of_month=row.get('DueDayOfMonth', ''), funding_source=row.get('FundingSource', 'ออมสิน'), start_next_day=safe_bool(row.get('StartNextDay')), is_locked_interest=safe_bool(row.get('IsLockedInterest')), locked_interest_amount=safe_float(row.get('LockedInterestAmount')) ) db.session.add(new_tx) count_success += 1 db.session.commit() print(f"Import CSV สำเร็จทั้งหมด {count_success} รายการ!") except Exception as e: db.session.rollback() print("Import CSV Error:", e) return redirect(url_for('receivables_dashboard'))
